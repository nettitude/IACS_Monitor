config system global
    set hostname FG-OT-PERIMETER-01
    set alias "OT Perimeter Firewall"
    set timezone 75
end
config system interface
    edit "port1"
        set vdom "root"
        set ip 10.0.1.1 255.255.255.0
        set allowaccess ping https ssh
        set type physical
        set zone "OT_ZONE"
        set alias "OT Control Network"
    next
    edit "port2"
        set vdom "root"
        set ip 10.0.2.1 255.255.255.0
        set type physical
        set zone "DMZ_ZONE"
        set alias "IT-OT DMZ"
    next
    edit "port3"
        set vdom "root"
        set ip 10.0.4.1 255.255.255.0
        set type physical
        set zone "IT_ZONE"
        set alias "IT Network"
    next
end
config system dns
    set primary 10.0.4.10
    set secondary 8.8.8.8
end

end

    edit "Host-PLC-02"

    edit "Host-HMI-01"

    next

        set type ipmask

config firewall policy
    edit 10
        set name "HMI-to-PLCs-ICS"
        set srcintf "port1"
        set dstintf "port1"
        set srcaddr "Host-HMI-01"
        set dstaddr "OT-PLCs"
        set service "ICS-Protocols"
        set action accept
        set logtraffic all
        set ips-sensor "OT-IPS-Profile"
        set comments "Owner: OT-Ops / ISA62443 SR 2.1 approved comms"
    next
    edit 20
        set name "JumpHost-to-OT-SSH"
        set srcintf "port2"
        set dstintf "port1"
        set srcaddr "Host-JumpServer"
        set dstaddr "Net-Engineering"
        set service "SSH"
        set action accept
        set logtraffic all
        set ips-sensor "OT-IPS-Profile"
        set comments "Owner: IT-Sec / Remote admin via jump host only - SR 1.13"
    next
    edit 30
        set name "OT-to-DMZ-Historian"

        set dstintf "port2"

        set dstaddr "Net-DMZ"

        set action accept

    next
    edit 98
        set name "No-Logging-Legacy"

        set dstintf "port1"
        set srcaddr "Net-Engineering"
        set dstaddr "Net-Engineering"

        set comments "Legacy rule - logging disabled"
    next
    edit 99
        set name "Deny-IT-to-OT-Direct"

        set dstintf "port1"

        set dstaddr "all"

        set action deny
        set logtraffic all

    edit 100
